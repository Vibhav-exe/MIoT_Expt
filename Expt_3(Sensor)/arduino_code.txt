// 1. Interfacing of Temperature Sensor (LM35)

#define sensorPin A0

void setup() {
  // Begin serial communication at 9600 baud rate
  Serial.begin(9600);
}

void loop() {
  // Get voltage reading from LM35
  int reading = analogRead(sensorPin);

  // Convert reading into voltage
  float voltage = reading * (5.0 / 1024.0);

  // Convert voltage into temperature in Celsius
  float temperatureC = voltage * 100;

  // Print temperature in Celsius
  Serial.print("Temperature: ");
  Serial.print(temperatureC);
  Serial.print(" C | ");

  // Convert Celsius into Fahrenheit
  float temperatureF = (temperatureC * 9.0 / 5.0) + 32.0;

  // Print temperature in Fahrenheit
  Serial.print(temperatureF);
  Serial.println(" F");

  delay(1000);
}

// 2. Interfacing of IR Sensor

int ledPin = 13;
int inputPin = 2;
int val = 0;

void setup() {
  pinMode(ledPin, OUTPUT);
  pinMode(inputPin, INPUT);
  Serial.begin(9600);
}

void loop() {
  val = digitalRead(inputPin);

  // Active LOW output
  if (val == HIGH) {
    Serial.println("Object Absent");
    digitalWrite(ledPin, LOW);
  }
  else {
    Serial.println("Object Present");
    digitalWrite(ledPin, HIGH);
  }
}

// 3. Interfacing of Ultrasonic Sensor (HC-SR04)

const int trigPin = 9;
const int echoPin = 2;

// Defining variables
long duration;
int distance;

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  Serial.begin(9600);
}

void loop() {
  // Clear the trigger pin
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  // Send 10 microsecond trigger pulse
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  // Read echo response
  duration = pulseIn(echoPin, HIGH);

  // Calculate distance in cm
  distance = duration * 0.034 / 2;

  // Print distance
  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");
}
