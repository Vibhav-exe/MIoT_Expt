from gpiozero import PWMOutputDevice, DigitalOutputDevice
from time import sleep

# Configure GPIO pins
enable = PWMOutputDevice(18)
in3 = DigitalOutputDevice(23)
in4 = DigitalOutputDevice(24)

# Clockwise rotation
in3.on()
in4.off()

# Increase speed
for i in range(0, 101):
    enable.value = i / 100
    sleep(0.02)

# Decrease speed
for i in range(100, -1, -1):
    enable.value = i / 100
    sleep(0.02)

# Stop motor
enable.off()
sleep(1)

# Anticlockwise rotation
in3.off()
in4.on()

enable.value = 0.7
sleep(3)

# Stop motor
enable.off()
